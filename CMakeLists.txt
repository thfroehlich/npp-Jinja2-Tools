cmake_minimum_required(VERSION 3.24)
project(Jinja2Tools VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Jinja2Tools is Windows-only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JINJA2TOOLS_BUILD_TESTS "Build tests" ON)
option(JINJA2TOOLS_BUILD_HELPER "Build Python helper" ON)
option(JINJA2TOOLS_PACKAGE_HELPER "Package Python helper" ON)
include(CTest)
include(cmake/BuildHelper.cmake)
add_subdirectory(plugin)
if(BUILD_TESTING AND JINJA2TOOLS_BUILD_TESTS)
  add_subdirectory(tests)
endif()
include(cmake/Packaging.cmake)
